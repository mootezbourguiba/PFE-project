\chapter{Final checklist before submission}

\section{Structure and coherence}

\begin{itemize}
    \item \textbf{✓} Front matter is complete.
    \item \textbf{✓} The introduction announces the problem statement, objectives and plan.
    \item \textbf{✓} Each chapter has a clear internal logic.
    \item \textbf{✓} The conclusion explicitly answers the objectives.
\end{itemize}

\section{Scientific quality}

\begin{itemize}
    \item \textbf{✓} Choices are justified and compared.
    \item \textbf{✓} Results are validated through an explicit protocol.
    \item \textbf{✓} Limitations and threats to validity are discussed.
    \item \textbf{✓} The report contains at least 25 references, with a majority of indexed scientific sources.
\end{itemize}

\section{Form and integrity}

\begin{itemize}
    \item \textbf{✓} Figures and tables are numbered and cited in the text.
    \item \textbf{✓} SI units are respected.
    \item \textbf{✓} The bibliography follows a homogeneous style (IEEE recommended).
    \item \textbf{✓} The target similarity rate remains < 15\%.
    \item \textbf{✓} Generative AI usage is declared if applicable.
\end{itemize}

\section{Files to submit}

\begin{itemize}
    \item Final PDF of the report.
    \item LaTeX source files (if required by the program).
    \item Similarity report.
    \item Oral defense slides.
    \item Technical appendices or complementary deliverables if required.
\end{itemize}
